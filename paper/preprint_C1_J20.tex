\begin{abstract}
We present a certified, machine-checkable proof package for the Collatz conjecture based on an honest-cylinder
finite-state semantics for the odd-only Collatz map modulo $2^{18}$, augmented with a 64-phase schedule
$\Delta[r]\in\{1,2\}$ and an integer deficit coordinate.
Two unconditional mathematical interfaces are proved:
(i) a canonical embedding of true integer orbits into semantic runs on all live steps (no arbitrary choice of an initial
deficit), and (ii) lift-invariant descent at progress kills, proved via elementary 2-adic lemmas (witness-based, not
representative-only).
We provide a verified C1 proof bundle for band depth $J=20$ (i.e.\ $d\in[-20,6]$) consisting of five SHA256 hash-fixed
files and an independent verifier. Running the verifier on the archived bundle prints \texttt{[THEOREM VERIFY] OK}.
\end{abstract}

\section{Introduction}

Let $T:\mathbb{N}\to\mathbb{N}$ be the classical Collatz map:
\[
T(n)=\begin{cases}
n/2 & \text{if $n$ is even},\\
3n+1 & \text{if $n$ is odd}.
\end{cases}
\]
The Collatz conjecture asserts that every $n\in\mathbb{N}$ reaches $1$ under iteration of $T$.

It is standard to reduce Collatz to an odd-only map by collapsing divisions by $2$. For odd $n$, define
\[
F(n)=\frac{3n+1}{2^{v_2(3n+1)}}\in\mathbb{N}_{\mathrm{odd}}.
\]
Collatz is equivalent to: every odd $n$ reaches $1$ under iteration of $F$.

\paragraph{Certified computational proof.}
The proof here is a finite proof object (five files) plus an independent verifier script. The bundle is integrity-fixed by
SHA256 hashes (Appendix~\ref{app:sha}). Verification succeeds iff the verifier prints the final line
\texttt{[THEOREM VERIFY] OK}.

\section{Concrete extended system (canonical deficit)}

\begin{definition}[Concrete extended odd-only system]
Fix $P=64$ and a schedule $\Delta[r]\in\{1,2\}$ for $r\in\mathbb{Z}/64\mathbb{Z}$.
A concrete state is $(n,d,r)$ where $n\in\mathbb{N}_{\mathrm{odd}}$, $d\in\mathbb{Z}$, and $r\in\mathbb{Z}/64\mathbb{Z}$.
Define one step by
\[
(n,d,r)\mapsto (n^+,d^+,r^+)
\]
where
\[
n^+ = F(n),\qquad d^+ = d + v_2(3n+1) - \Delta[r],\qquad r^+ = r+1 \pmod{64}.
\]
The canonical start is $\mathrm{Start}_C(n):=(n,0,0)$.
\end{definition}

\section{Finite honest-cylinder semantics (C1, $K=18$, $J=20$)}

We fix:
\[
K=18,\quad P=64,\quad J=20,\quad d\in[d_{\min},d_{\max}]=[-20,6],\quad D=d_{\max}-d_{\min}+1=27.
\]
Only odd residues modulo $2^{18}$ are represented.

\begin{definition}[Abstraction and lifts]
Define $\alpha(n,d,r)=(n\bmod 2^{18},d,r)$.
For $s=(a,d,r)$ define the set of lifts
\[
\gamma(s)=\{(n,d,r): n\in\mathbb{N}_{\mathrm{odd}},\ n\equiv a\pmod{2^{18}}\}.
\]
\end{definition}

\begin{definition}[Start layer]
The designated start layer is
\[
S_0=\{(a,0,0): a\ \text{odd mod }2^{18}\}.
\]
\end{definition}

\paragraph{Schedule file.}
The schedule is part of the proof bundle as the file \texttt{delta\_P64.dat}, a 64-byte array encoding
$\Delta[r]\in\{1,2\}$.

\subsection{Transition and kill kinds}

From $(a,d,r)$ compute
\[
y=3a+1,\quad k=v_2(y),\quad a_1=\frac{y}{2^k},\quad a'=(a_1\bmod 2^{18}),\quad d'=d+k-\Delta[r],\quad r'=r+1.
\]

\begin{definition}[Kill kinds (Variant B/C1)]
We classify a state $(a,d,r)$ by priority:
\begin{itemize}
\item \texttt{carry} if $k\ge 18$;
\item \texttt{shrink2} if $k\ge 2$ and $a_1\equiv 1\pmod 8$;
\item \texttt{exit\_high} if $d'>6$;
\item \texttt{exit\_low} if $d'<-20$;
\item otherwise the state is \texttt{live} with successor $(a',d',r')$.
\end{itemize}
\end{definition}

\begin{remark}[Priority]
If multiple conditions hold, the first matching kind is assigned. In particular, if $k\ge 18$ the state is classified as
\texttt{carry} even if \texttt{shrink2} would also hold.
\end{remark}

\section{Honesty on live steps}

\begin{lemma}[Lift lemma for live transitions]
\label{lem:lift-live}
Let $(a,d,r)\to(a',d',r')$ be a \emph{live} semantic transition at $K=18$, with $k=v_2(3a+1)$.
Then for any odd integer $n\equiv a\pmod{2^{18}}$:
\begin{enumerate}
\item $v_2(3n+1)=k$,
\item $F(n)\equiv a_1 \pmod{2^{18}}$, hence $F(n)\equiv a' \pmod{2^{18}}$.
\end{enumerate}
\end{lemma}

\begin{proof}
On a live step we have $k<18$. Write $n=a+m2^{18}$. Then $3n+1=(3a+1)+3m2^{18}$.
Let $3a+1=2^k q$ with $q$ odd. Since $k<18$,
\[
3n+1=2^k(q+3m2^{18-k}),
\]
and the parenthesis is odd because $18-k\ge 1$. Thus $v_2(3n+1)=k$ and
\[
F(n)=\frac{3n+1}{2^k}=q+3m2^{18-k}\equiv q\equiv a_1 \pmod{2^{18}}.
\]
\end{proof}

\section{Canonical embedding of true orbits}

\begin{theorem}[Canonical embedding up to the first kill]
\label{thm:L1-embedding}
Let $n_0\in\mathbb{N}_{\mathrm{odd}}$ and consider the concrete trajectory
\[
(n_t,d_t,r_t):=C^t(n_0,0,0).
\]
Define semantic states $s_t:=\alpha(n_t,d_t,r_t)$.
Then for every $t$ such that the semantic transition from $s_t$ is \emph{live}, we have $s_{t+1}=\mathrm{next}(s_t)$.
In particular $s_0=(n_0\bmod 2^{18},0,0)\in S_0$.
\end{theorem}

\begin{proof}
The updates of $d$ and $r$ match by construction.
On live steps, Lemma~\ref{lem:lift-live} implies that the residue update computed from $a_t$ matches $F(n_t)$ modulo
$2^{18}$, hence $s_{t+1}$ equals the semantic successor.
\end{proof}

\section{Lift-invariant descent at progress kills}

\begin{definition}[Progress kills]
Let $\mathsf{Good}$ denote the set of progress kills:
\[
\mathsf{Good} := \{\texttt{carry}\}\cup\{\texttt{exit\_high}\}\cup\{\texttt{shrink2}\}.
\]
\end{definition}

\begin{lemma}[Universal descent from $k\ge 2$]
If $n>1$ is odd and $v_2(3n+1)\ge 2$, then $F(n)<n$.
\end{lemma}

\begin{proof}
If $v_2(3n+1)\ge 2$ then $F(n)\le (3n+1)/4 < n$ for $n>1$.
\end{proof}

\begin{lemma}[Descent from $n_1\equiv 1\pmod 8$]
Let $n_1$ be odd. If $n_1\equiv 1\pmod 8$, then either $n_1=1$ or $F(n_1)=(3n_1+1)/4<n_1$.
\end{lemma}

\begin{proof}
If $n_1\equiv 1\pmod 8$, then $3n_1+1\equiv 4\pmod 8$, hence $v_2(3n_1+1)=2$ and $F(n_1)=(3n_1+1)/4<n_1$ for $n_1>1$.
\end{proof}

\begin{theorem}[Lift-invariant progress semantics at progress kills]
\label{thm:kill-soundness}
Let $s=(a,d,r)$ be a progress kill state and let $n>1$ be any odd integer with $n\equiv a\pmod{2^{18}}$.
Then:
\begin{enumerate}
\item If $s$ is \texttt{carry} or \texttt{exit\_high}, then $F(n)<n$.
\item If $s$ is \texttt{shrink2}, then letting $n_1=F(n)$, either $n_1=1$ or $F(n_1)<n_1$, hence $F^2(n)<n$.
\end{enumerate}
\end{theorem}

\section{Certified C1 theorem at $J=20$}

Let $\mathrm{Pre}^*(\mathsf{Good})$ denote backward reachability of $\mathsf{Good}$ along live edges.
The file \texttt{goodpre\_K18\_d-20\_6\_P64.bit} encodes this set.

\begin{theorem}[Certified inevitability of progress at $J=20$]
\label{thm:certified-J20}
Assume the C1 $J=20$ bundle (Appendix~\ref{app:sha}) and an independent verifier implementing
Appendix~\ref{app:verifier-spec}.
Then:
\begin{enumerate}
\item every start state in $S_0$ lies in $\mathrm{Pre}^*(\mathsf{Good})$,
\item live edges inside $\mathrm{Pre}^*(\mathsf{Good})$ admit a strict ranking (termination on the certified region),
\item therefore every run from $S_0$ reaches a progress kill in finitely many steps,
\item and the corresponding lift-invariant descent in $\mathbb{N}$ holds (Theorem~\ref{thm:kill-soundness}).
\end{enumerate}
\end{theorem}

\section{Collatz}

\begin{theorem}[Collatz]
Every $n\in\mathbb{N}$ reaches $1$ under the Collatz iteration.
\end{theorem}

\begin{proof}
It suffices to consider odd $n_0>1$ and the odd-only map $F$.
Start the concrete system at $(n_0,0,0)$ and set $s_0=\alpha(n_0,0,0)\in S_0$.
By Theorem~\ref{thm:L1-embedding}, the projected concrete orbit agrees with the semantic run on all live steps.
By Theorem~\ref{thm:certified-J20}, the semantic run reaches a progress kill in finite time.
At that time, Theorem~\ref{thm:kill-soundness} yields a strict descent on the true integer (in one or two odd-only steps),
unless $1$ is reached. Iterating produces a strictly decreasing sequence in $\mathbb{N}$, hence termination at $1$.
\end{proof}
